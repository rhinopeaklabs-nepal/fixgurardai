# Upload this file to your Web Hosting public_html as ".htaccess"
# (with the leading dot). Without it, /r/<token> share links and /prompts
# return 404 on refresh, because Apache looks for a folder that isn't there.

<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteBase /

  # Real files and folders are served as-is.
  RewriteCond %{REQUEST_FILENAME} -f [OR]
  RewriteCond %{REQUEST_FILENAME} -d
  RewriteRule ^ - [L]

  # Everything else is handled by the single-page app.
  RewriteRule . /index.html [L]
</IfModule>

# Hashed asset filenames change on every build, so they can cache forever.
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType text/css              "access plus 1 year"
  ExpiresByType application/javascript "access plus 1 year"
  ExpiresByType image/svg+xml          "access plus 1 month"
  ExpiresDefault                       "access plus 1 hour"
</IfModule>

# index.html must never be cached, or people keep loading the old app.
<FilesMatch "index\.html$">
  <IfModule mod_headers.c>
    Header set Cache-Control "no-cache, must-revalidate"
  </IfModule>
</FilesMatch>

<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>
